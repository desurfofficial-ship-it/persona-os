   119→   119→function timeAgo(iso?: string | null): string {
   120→   120→  if (!iso) return "never";
   121→   121→  const diff = Date.now() - new Date(iso).getTime();
   122→   122→  const mins = Math.floor(diff / 60000);
   123→   123→  if (mins < 1) return "just now";
   124→   124→  if (mins < 60) return `${mins}m ago`;
   125→   125→  const hours = Math.floor(mins / 60);
   126→   126→  if (hours < 24) return `${hours}h ago`;
   127→   127→  return `${Math.floor(hours / 24)}d ago`;
   128→   128→}
   129→   129→
   130→   130→/** Human label for a FUTURE timestamp — timeAgo says "just now" for those. */
   131→   131→function timeUntil(iso?: string | null): string {
   132→   132→  if (!iso) return "not scheduled";
   133→   133→  const diff = new Date(iso).getTime() - Date.now();
   134→   134→  if (diff <= 0) return "due now";
   135→   135→  const mins = Math.floor(diff / 60000);
   136→   136→  if (mins < 60) return `due in ${mins}m`;
   137→   137→  const hours = Math.floor(mins / 60);
   138→   138→  if (hours < 24) return `due in ${hours}h`;
   139→   139→  return `due in ${Math.floor(hours / 24)}d`;
   140→   140→}

   166→   166→    let alive = true;

   168→   168→      const { data } = await supabase.auth.getUser();

   170→   170→      if (!data.user) {
   171→   171→        router.push("/login");
   172→   172→        return;
   173→   173→      }
   174→   174→      setAuthReady(true);
   175→   175→      try {
   176→   176→        const res = await authedFetch("/api/personas");
   177→   177→        const json = (await res.json()) as { personas?: Persona[] };
   178→   178→        const list = Array.isArray(json.personas) ? json.personas : [];
   179→   179→        if (!alive) return;
   180→   180→        setPersonas(list);
   181→   181→        if (list.length > 0) setSelectedId((prev) => prev || list[0].id);
   182→   182→      } catch {
   183→   183→        if (alive) setPersonas([]);
   184→   184→      }
   185→   185→    })();

   187→   187→      alive = false;
   188→   188→    };
   189→   189→  }, [router]);
   190→   190→
   191→   191→  const activePersona = useMemo(
   192→   192→    () => personas.find((p) => p.id === selectedId) || null,
   193→   193→    [personas, selectedId]

   201→   201→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   202→   202→      ...(selectedId ? { "x-persona-id": selectedId } : {}),
   203→   203→    };
   204→   204→  }, [selectedId]);
   205→   205→
   206→   206→  if (!authReady) {
   207→   207→    return (
   208→   208→      <div className="min-h-screen flex items-center justify-center text-sm" style={{ background: CREAM, color: CHARCOAL }}>
   209→   209→        Loading agent…

   211→   211→    );
   212→   212→  }
   213→   213→
   214→   214→  return (
   215→   215→    <CopilotKit runtimeUrl="/api/copilotkit" headers={buildHeaders}>
   216→   216→      <AgentWorkspace
   217→   217→        personas={personas}
   218→   218→        selectedId={selectedId}
   219→   219→        onSelect={setSelectedId}
   220→   220→        activePersona={activePersona}
   221→   221→      />
   222→   222→    </CopilotKit>
   223→   223→  );
