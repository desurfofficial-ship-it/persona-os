    81→function hostOf(url: string): string {

    89→function timeAgo(iso?: string | null): string {
    90→  if (!iso) return "never";
    91→  const diff = Date.now() - new Date(iso).getTime();
    92→  const mins = Math.floor(diff / 60000);
    93→  if (mins < 1) return "just now";
    94→  if (mins < 60) return `${mins}m ago`;
    95→  const hours = Math.floor(mins / 60);
    96→  if (hours < 24) return `${hours}h ago`;
    97→  return `${Math.floor(hours / 24)}d ago`;
    98→}
    99→
   100→/** Human label for a FUTURE timestamp — timeAgo says "just now" for those. */
   101→function timeUntil(iso?: string | null): string {
   102→  if (!iso) return "not scheduled";
   103→  const diff = new Date(iso).getTime() - Date.now();
   104→  if (diff <= 0) return "due now";
   105→  const mins = Math.floor(diff / 60000);
   106→  if (mins < 60) return `due in ${mins}m`;
   107→  const hours = Math.floor(mins / 60);
   108→  if (hours < 24) return `due in ${hours}h`;
   109→  return `due in ${Math.floor(hours / 24)}d`;
   110→}

   112→/**
   113→ * The composer tags every prompt `[Model:…] [Type:…] [Persona:…]` so the wire
   114→ * transcript shows exactly what the agent was asked — but users should never
   115→ * see the raw tags (or the persona UUID). Strip them for display only.
   116→ */
   117→function stripMeta(text: string): string {
   118→  return text
   119→    .replace(/\s*\[(?:Model|Type|Persona):[^\]]*\]/g, "")
   120→    .trim();
   121→}
   122→
   123→/** Collapse full UUIDs to 8 chars — tool results read better in small cards. */
   124→function shortUuids(text: string): string {
   125→  return text.replace(/\b[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}\b/gi, (m) => `${m.slice(0, 8)}…`);
   126→}

   128→export default function AgentPage() {
   129→  const router = useRouter();
   130→  const [authReady, setAuthReady] = useState(false);
   131→  const [personas, setPersonas] = useState<Persona[]>([]);
   132→  const [selectedId, setSelectedId] = useState<string>("");
   133→
   134→  // Auth gate + persona load from /api/personas.
   135→  useEffect(() => {
   136→    let alive = true;
   137→    (async () => {
   138→      const { data } = await supabase.auth.getUser();
   139→      if (!alive) return;
   140→      if (!data.user) {
   141→        router.push("/login");
   142→        return;
   143→      }
   144→      setAuthReady(true);
   145→      try {
   146→        const res = await authedFetch("/api/personas");
   147→        const json = (await res.json()) as { personas?: Persona[] };
   148→        const list = Array.isArray(json.personas) ? json.personas : [];
   149→        if (!alive) return;
   150→        setPersonas(list);
   151→        if (list.length > 0) setSelectedId((prev) => prev || list[0].id);
   152→      } catch {
   153→        if (alive) setPersonas([]);
   154→      }
   155→    })();
   156→    return () => {
   157→      alive = false;
   158→    };
   159→  }, [router]);
   160→
   161→  const activePersona = useMemo(
   162→    () => personas.find((p) => p.id === selectedId) || null,
   163→    [personas, selectedId]
   164→  );
   165→
   166→  // Headers callback — evaluated per runtime request by CopilotKit; a plain
   167→  // closure over state (no ref, no render-time writes).
   168→  const buildHeaders = useCallback((): Record<string, string> => {
   169→    const token = getSessionToken();
   170→    return {
   171→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   172→      ...(selectedId ? { "x-persona-id": selectedId } : {}),
   173→    };
   174→  }, [selectedId]);
   175→
   176→  if (!authReady) {
   177→    return (
   178→      <div className="min-h-screen flex items-center justify-center text-sm" style={{ background: CREAM, color: CHARCOAL }}>
   179→        Loading agent…
   180→      </div>
   181→    );
   182→  }
   183→
   184→  return (
   185→    <CopilotKit runtimeUrl="/api/copilotkit" headers={buildHeaders}>
   186→      <AgentWorkspace
   187→        personas={personas}
   188→        selectedId={selectedId}
   189→        onSelect={setSelectedId}
   190→        activePersona={activePersona}
   191→      />
   192→    </CopilotKit>
   193→  );

   509→function CopilotChatPane({
   510→  visibleMessages,
   511→  isLoading,
   512→  type,
   513→  model,
   514→  personaName,
   515→}: {
   516→  visibleMessages: unknown[];
   517→  isLoading: boolean;
   518→  type: AgentContentType;
   519→  model: string;
   520→  personaName: string | null;
   521→}) {
   522→  const scrollRef = useRef<HTMLDivElement | null>(null);
   523→
   524→  const items = useMemo(() => {
   525→    const source = Array.isArray(visibleMessages) ? visibleMessages : [];
   526→    return source
   527→      .map((m) => m as { id?: string; type?: string; role?: string; content?: unknown; name?: string; toolCalls?: Array<{ id?: string; function?: { name?: string; arguments?: string } }>; result?: unknown })
   528→      .filter((m) => {
   529→        // AG-UI shapes (role user/assistant/tool) and legacy class shapes.
   530→        if (m.type === "ActionExecutionMessage") return true;
   531→        if (m.type === "ResultMessage") return true;
   532→        if (m.role === "user" || m.role === "assistant" || m.role === "tool") {
   533→          const hasText = typeof m.content === "string" && m.content.trim().length > 0;
   534→          return hasText || (Array.isArray(m.toolCalls) && m.toolCalls.length > 0);
   535→        }
   536→        return false;
   537→      })
   538→      .slice(-60);
   539→  }, [visibleMessages]);
   540→
   541→  useEffect(() => {
   542→    const el = scrollRef.current;
   543→    if (el) el.scrollTop = el.scrollHeight;
   544→  }, [items.length, isLoading]);
   545→
   546→  return (
   547→    <div className="flex-1 flex flex-col min-h-[50vh] xl:min-h-0">
   548→      <div ref={scrollRef} className="flex-1 overflow-y-auto px-4 py-5 space-y-3">
   549→        <div className="rounded-xl border p-4 text-xs leading-relaxed" style={{ borderColor: "#EDE3D6", background: "#FFFFFF" }}>
   550→          <p className="font-serif text-sm mb-1" style={{ color: CHARCOAL }}>
   551→            Persona Agent
   552→          </p>
   553→          <p style={{ color: "#8A8177" }}>
   554→            {personaName
   555→              ? `Generating ${type} using ${model} for ${personaName}. Ask for research, a week of posts, scheduling or vault saves — the agent acts, not just suggests.`
   556→              : "Pick a persona on the left, then ask for research, posts, scheduling or vault saves."}
   557→          </p>
   558→        </div>
   559→
   560→        {items.map((m, i) => {
   561→          const toolCalls = Array.isArray(m.toolCalls) ? m.toolCalls : [];
   562→          if (toolCalls.length > 0) {
   563→            return (
   564→              <div key={m.id || i} className="flex flex-col gap-1 items-start">
   565→                {toolCalls.map((tc, j) => (
   566→                  <div key={tc.id || j} className="text-[11px] rounded-lg px-3 py-1.5 inline-block" style={{ background: "#FFF3EC", color: ACCENT }}>
   567→                    🔧 {tc.function?.name || "tool"}…
   568→                  </div>
   569→                ))}
   570→              </div>
   571→            );
   572→          }
   573→          if (m.type === "ActionExecutionMessage") {
   574→            return (
   575→              <div key={m.id || i} className="text-[11px] rounded-lg px-3 py-1.5 inline-block" style={{ background: "#FFF3EC", color: ACCENT }}>
   576→                🔧 {m.name}…
   577→              </div>
   578→            );
   579→          }
   580→          if (m.type === "ResultMessage" || m.role === "tool") {
   581→            const result = String(m.result ?? (typeof m.content === "string" ? m.content : "")).slice(0, 200);
   582→            return (
   583→              <div key={m.id || i} className="text-[11px] rounded-lg px-3 py-1.5" style={{ background: "#EDF5EE", color: "#2F6B37" }}>
   584→                ✓ {result}
   585→              </div>
   586→            );
   587→          }
   588→          const isUser = m.role === "user";
   589→          // Wire tags are for the runtime + audit; humans get the clean prompt.
   590→          const content = isUser ? stripMeta(typeof m.content === "string" ? m.content : "") : shortUuids(typeof m.content === "string" ? m.content : "");
   591→          return (
   592→            <div key={m.id || i} className={`flex ${isUser ? "justify-end" : "justify-start"}`}>
   593→              <div
   594→                className={`max-w-[85%] rounded-2xl px-4 py-2.5 text-sm leading-relaxed whitespace-pre-wrap break-words ${isUser ? "text-white" : ""}`}
   595→                style={{
   596→                  background: isUser ? ACCENT : "#FFFFFF",
   597→                  color: isUser ? "#FFFFFF" : CHARCOAL,
   598→                  border: isUser ? "none" : "1px solid #EDE3D6",
   599→                }}
   600→              >
   601→                {content}
   602→              </div>
   603→            </div>
   604→          );
   605→        })}
   606→
   607→        {isLoading && (
   608→          <div className="text-xs px-2" style={{ color: "#8A8177" }} aria-live="polite">
   609→            Working…
   610→          </div>
   611→        )}
   612→      </div>
   613→    </div>
   614→  );
   615→}

   621→function GoalsPanel({

   633→  const [alerts, setAlerts] = useState<AlertRow[]>([]);
   634→  const [open, setOpen] = useState(false);

   641→  const [togglingId, setTogglingId] = useState<string | null>(null);
   642→  const [refresh, setRefresh] = useState(0);
   643→
   644→  useEffect(() => {
   645→    setGoalPersonaId((prev) => prev || activePersona?.id || "");
   646→  }, [activePersona?.id]);
   647→
   648→  useEffect(() => {
   649→    let alive = true;
   650→    (async () => {
   651→      try {
   652→        const res = await authedFetch("/api/goals");
   653→        if (!res.ok) return;
   654→        const json = (await res.json()) as { goals?: GoalRow[]; alerts?: AlertRow[] };
   655→        if (!alive) return;
   656→        setGoals(Array.isArray(json.goals) ? json.goals : []);
   657→        setAlerts(Array.isArray(json.alerts) ? json.alerts : []);
   658→      } catch {
   659→        // Panel stays empty on failure — never blocks the page.
   660→      }
   661→    })();

   663→      alive = false;

   672→        method: "POST",
   673→        headers: { "Content-Type": "application/json" },
   674→        body: JSON.stringify({
   675→          goalTitle: goalTitle.trim(),
   676→          recurrence,
   677→          checkUrl: checkUrl.trim(),

   679→        }),
   680→      });
   681→      const json = (await res.json()) as Record<string, unknown>;
   682→      if (!res.ok) throw new Error((json.error as string) || "Failed");
   683→      setGoalTitle("");
   684→      pushNotice(`Goal created → ${goalTitle.trim()}`);
   685→      setRefresh((n) => n + 1);
   686→    } catch (err) {
   687→      pushNotice(`Goal failed: ${err instanceof Error ? err.message : "error"}`);
   688→    } finally {
   689→      setSaving(false);
   690→    }
   691→  };
   692→
   693→  const runCheck = async (goalId: string) => {

   695→    try {
   696→      const res = await authedFetch("/api/goals/check", {
   697→        method: "POST",
   698→        headers: { "Content-Type": "application/json" },
   699→        body: JSON.stringify({ goalId }),
   700→      });
   701→      const json = (await res.json()) as {
   702→        results?: Array<{ outcome?: string; detail?: string }>;
   703→      };
   704→      const first = json.results?.[0];
   705→      if (!first) pushNotice("Check ran → goal not due yet (its schedule is respected).");
   706→      else if (first.outcome === "generated") pushNotice("Check ran → page changed → auto-draft created ✓");

   711→      setRefresh((n) => n + 1);

   713→      pushNotice("Check failed to run.");
   714→    } finally {
   715→      setCheckingId(null);

   793→          <p className="text-[10px]" style={{ color: "#8A8177" }}>
   794→            Goals &amp; Tracking — recurring page checks that auto-generate drafts

   995→        </div>
