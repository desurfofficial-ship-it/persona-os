    79→function hostOf(url: string): string {
    80→  try {
    81→    return new URL(url).host.replace(/^www\./, "");
    82→  } catch {
    83→    return url;
    84→  }

    87→function timeAgo(iso?: string | null): string {
    88→  if (!iso) return "never";
    89→  const diff = Date.now() - new Date(iso).getTime();
    90→  const mins = Math.floor(diff / 60000);
    91→  if (mins < 1) return "just now";
    92→  if (mins < 60) return `${mins}m ago`;
    93→  const hours = Math.floor(mins / 60);
    94→  if (hours < 24) return `${hours}h ago`;
    95→  return `${Math.floor(hours / 24)}d ago`;
    96→}

    98→export default function AgentPage() {
    99→  const router = useRouter();
   100→  const [authReady, setAuthReady] = useState(false);
   101→  const [personas, setPersonas] = useState<Persona[]>([]);
   102→  const [selectedId, setSelectedId] = useState<string>("");
   103→
   104→  // Auth gate + persona load from /api/personas.
   105→  useEffect(() => {
   106→    let alive = true;
   107→    (async () => {
   108→      const { data } = await supabase.auth.getUser();
   109→      if (!alive) return;
   110→      if (!data.user) {
   111→        router.push("/login");
   112→        return;
   113→      }
   114→      setAuthReady(true);
   115→      try {
   116→        const res = await authedFetch("/api/personas");
   117→        const json = (await res.json()) as { personas?: Persona[] };
   118→        const list = Array.isArray(json.personas) ? json.personas : [];
   119→        if (!alive) return;
   120→        setPersonas(list);
   121→        if (list.length > 0) setSelectedId((prev) => prev || list[0].id);
   122→      } catch {
   123→        if (alive) setPersonas([]);
   124→      }
   125→    })();
   126→    return () => {
   127→      alive = false;
   128→    };
   129→  }, [router]);
   130→
   131→  const activePersona = useMemo(
   132→    () => personas.find((p) => p.id === selectedId) || null,
   133→    [personas, selectedId]
   134→  );
   135→
   136→  // Headers callback — evaluated per runtime request by CopilotKit; a plain
   137→  // closure over state (no ref, no render-time writes).
   138→  const buildHeaders = useCallback((): Record<string, string> => {
   139→    const token = getSessionToken();
   140→    return {
   141→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   142→      ...(selectedId ? { "x-persona-id": selectedId } : {}),
   143→    };
   144→  }, [selectedId]);
   145→
   146→  if (!authReady) {
   147→    return (
   148→      <div className="min-h-screen flex items-center justify-center text-sm" style={{ background: CREAM, color: CHARCOAL }}>
   149→        Loading agent…
   150→      </div>
   151→    );
   152→  }
   153→
   154→  return (
   155→    <CopilotKit runtimeUrl="/api/copilotkit" headers={buildHeaders}>
   156→      <AgentWorkspace
   157→        personas={personas}
   158→        selectedId={selectedId}
   159→        onSelect={setSelectedId}
   160→        activePersona={activePersona}
   161→      />
   162→    </CopilotKit>
   163→  );
   164→}

   468→// ---------------------------------------------------------------------------
   469→// CopilotChatPane — headless rendering of the agent conversation.
   470→// Same stream as the stock <CopilotChat/> (useCopilotChat), styled to this
   471→// page: cream background, charcoal text, coral agent accents. Text messages,
   472→// tool-call chips (saveToDrafts, saveToVault, createGoal, scheduleContent…)
   473→// and tool results are all visible so auto-saves are never a black box.
   474→// ---------------------------------------------------------------------------
   475→
   476→function CopilotChatPane({
   477→  visibleMessages,
   478→  isLoading,
   479→  type,
   480→  model,
   481→  personaName,
   482→}: {
   483→  visibleMessages: unknown[];
   484→  isLoading: boolean;
   485→  type: AgentContentType;
   486→  model: string;
   487→  personaName: string | null;
   488→}) {
   489→  const scrollRef = useRef<HTMLDivElement | null>(null);
   490→
   491→  const items = useMemo(() => {
   492→    const source = Array.isArray(visibleMessages) ? visibleMessages : [];
   493→    return source
   494→      .map((m) => m as { id?: string; type?: string; role?: string; content?: unknown; name?: string; toolCalls?: Array<{ id?: string; function?: { name?: string; arguments?: string } }>; result?: unknown })
   495→      .filter((m) => {
   496→        // AG-UI shapes (role user/assistant/tool) and legacy class shapes.
   497→        if (m.type === "ActionExecutionMessage") return true;
   498→        if (m.type === "ResultMessage") return true;
   499→        if (m.role === "user" || m.role === "assistant" || m.role === "tool") {
   500→          const hasText = typeof m.content === "string" && m.content.trim().length > 0;
   501→          return hasText || (Array.isArray(m.toolCalls) && m.toolCalls.length > 0);
   502→        }
   503→        return false;
   504→      })
   505→      .slice(-60);
   506→  }, [visibleMessages]);
   507→
   508→  useEffect(() => {
   509→    const el = scrollRef.current;
   510→    if (el) el.scrollTop = el.scrollHeight;
   511→  }, [items.length, isLoading]);
   512→
   513→  return (
   514→    <div className="flex-1 flex flex-col min-h-[50vh] xl:min-h-0">
   515→      <div ref={scrollRef} className="flex-1 overflow-y-auto px-4 py-5 space-y-3">
   516→        <div className="rounded-xl border p-4 text-xs leading-relaxed" style={{ borderColor: "#EDE3D6", background: "#FFFFFF" }}>
   517→          <p className="font-serif text-sm mb-1" style={{ color: CHARCOAL }}>
   518→            Persona Agent
   519→          </p>
   520→          <p style={{ color: "#8A8177" }}>
   521→            {personaName
   522→              ? `Generating ${type} using ${model} for ${personaName}. Ask for research, a week of posts, scheduling or vault saves — the agent acts, not just suggests.`
   523→              : "Pick a persona on the left, then ask for research, posts, scheduling or vault saves."}
   524→          </p>
   525→        </div>
   526→
   527→        {items.map((m, i) => {
   528→          const toolCalls = Array.isArray(m.toolCalls) ? m.toolCalls : [];
   529→          if (toolCalls.length > 0) {
   530→            return (
   531→              <div key={m.id || i} className="flex flex-col gap-1 items-start">
   532→                {toolCalls.map((tc, j) => (
   533→                  <div key={tc.id || j} className="text-[11px] rounded-lg px-3 py-1.5 inline-block" style={{ background: "#FFF3EC", color: ACCENT }}>
   534→                    🔧 {tc.function?.name || "tool"}…
   535→                  </div>
   536→                ))}
   537→              </div>
   538→            );
   539→          }
   540→          if (m.type === "ActionExecutionMessage") {
   541→            return (
   542→              <div key={m.id || i} className="text-[11px] rounded-lg px-3 py-1.5 inline-block" style={{ background: "#FFF3EC", color: ACCENT }}>
   543→                🔧 {m.name}…
   544→              </div>
   545→            );
   546→          }
   547→          if (m.type === "ResultMessage" || m.role === "tool") {
   548→            const result = String(m.result ?? (typeof m.content === "string" ? m.content : "")).slice(0, 200);
   549→            return (
   550→              <div key={m.id || i} className="text-[11px] rounded-lg px-3 py-1.5" style={{ background: "#EDF5EE", color: "#2F6B37" }}>
   551→                ✓ {result}
   552→              </div>
   553→            );
   554→          }
   555→          const isUser = m.role === "user";
   556→          const content = typeof m.content === "string" ? m.content : "";
   557→          return (
   558→            <div key={m.id || i} className={`flex ${isUser ? "justify-end" : "justify-start"}`}>
   559→              <div
   560→                className={`max-w-[85%] rounded-2xl px-4 py-2.5 text-sm leading-relaxed whitespace-pre-wrap break-words ${isUser ? "text-white" : ""}`}
   561→                style={{
   562→                  background: isUser ? ACCENT : "#FFFFFF",
   563→                  color: isUser ? "#FFFFFF" : CHARCOAL,
   564→                  border: isUser ? "none" : "1px solid #EDE3D6",
   565→                }}
   566→              >
   567→                {content}
   568→              </div>
   569→            </div>
   570→          );
   571→        })}
   572→
   573→        {isLoading && (
   574→          <div className="text-xs px-2" style={{ color: "#8A8177" }} aria-live="polite">
   575→            Working…
   576→          </div>
   577→        )}
   578→      </div>
   579→    </div>
   580→  );
   581→}

   587→function GoalsPanel({
   588→  activePersona,
   589→  pushNotice,
   590→  refreshSignal,
   591→}: {
   592→  activePersona: Persona | null;
   593→  pushNotice: (text: string) => void;
   594→  refreshSignal: number;
   595→}) {
   596→  const [goals, setGoals] = useState<GoalRow[]>([]);
   597→  const [alerts, setAlerts] = useState<AlertRow[]>([]);
   598→  const [open, setOpen] = useState(false);
   599→  const [goalTitle, setGoalTitle] = useState("");
   600→  const [recurrence, setRecurrence] = useState<"daily" | "weekly">("weekly");
   601→  const [checkUrl, setCheckUrl] = useState("https://www.tiktok.com/tag/wellness");
   602→  const [goalPersonaId, setGoalPersonaId] = useState("");
   603→  const [saving, setSaving] = useState(false);
   604→  const [checkingId, setCheckingId] = useState<string | null>(null);
   605→  const [refresh, setRefresh] = useState(0);
   606→
   607→  useEffect(() => {
   608→    setGoalPersonaId((prev) => prev || activePersona?.id || "");
   609→  }, [activePersona?.id]);
   610→
   611→  useEffect(() => {
   612→    let alive = true;
   613→    (async () => {
   614→      try {
   615→        const res = await authedFetch("/api/goals");
   616→        if (!res.ok) return;
   617→        const json = (await res.json()) as { goals?: GoalRow[]; alerts?: AlertRow[] };
   618→        if (!alive) return;
   619→        setGoals(Array.isArray(json.goals) ? json.goals : []);
   620→        setAlerts(Array.isArray(json.alerts) ? json.alerts : []);
   621→      } catch {
   622→        // Panel stays empty on failure — never blocks the page.
   623→      }
   624→    })();
   625→    return () => {
   626→      alive = false;
   627→    };
   628→  }, [refresh, refreshSignal]);
   629→
   630→  const createGoal = async () => {
   631→    if (saving) return;
   632→    setSaving(true);
   633→    try {
   634→      const res = await authedFetch("/api/goals", {
   635→        method: "POST",
   636→        headers: { "Content-Type": "application/json" },
   637→        body: JSON.stringify({
   638→          goalTitle: goalTitle.trim(),
   639→          recurrence,
   640→          checkUrl: checkUrl.trim(),
   641→          persona_id: goalPersonaId || activePersona?.id,
   642→        }),
   643→      });
   644→      const json = (await res.json()) as Record<string, unknown>;
   645→      if (!res.ok) throw new Error((json.error as string) || "Failed");
   646→      setGoalTitle("");
   647→      pushNotice(`Goal created → ${goalTitle.trim()}`);
   648→      setRefresh((n) => n + 1);
   649→    } catch (err) {
   650→      pushNotice(`Goal failed: ${err instanceof Error ? err.message : "error"}`);
   651→    } finally {
   652→      setSaving(false);
   653→    }
   654→  };
   655→
   656→  const runCheck = async (goalId: string) => {
   657→    setCheckingId(goalId);
   658→    try {
   659→      const res = await authedFetch("/api/goals/check", {
   660→        method: "POST",
   661→        headers: { "Content-Type": "application/json" },
   662→        body: JSON.stringify({ goalId }),
   663→      });
   664→      const json = (await res.json()) as {
   665→        results?: Array<{ outcome?: string; detail?: string }>;
   666→      };
   667→      const first = json.results?.[0];
   668→      if (!first) pushNotice("Check ran → goal not due yet (its schedule is respected).");
   669→      else if (first.outcome === "generated") pushNotice("Check ran → page changed → auto-draft created ✓");
   670→      else if (first.outcome === "changed") pushNotice("Check ran → page changed ✓ (alert raised)");
   671→      else if (first.outcome === "baseline") pushNotice("Baseline recorded — next change will trigger.");
   672→      else if (first.outcome === "unchanged") pushNotice("Check ran → nothing new.");
   673→      else pushNotice(`Check ran → ${first.outcome || "error"}${first.detail ? ` (${first.detail})` : ""}`);
   674→      setRefresh((n) => n + 1);
   675→    } catch {
   676→      pushNotice("Check failed to run.");
   677→    } finally {
   678→      setCheckingId(null);
   679→    }
   680→  };
   681→
   682→  const removeGoal = async (goalId: string) => {
   683→    try {
   684→      await authedFetch(`/api/goals?id=${encodeURIComponent(goalId)}`, { method: "DELETE" });
   685→      setRefresh((n) => n + 1);
   686→    } catch {
   687→      pushNotice("Could not remove the goal.");
   688→    }
   689→  };
   690→
   691→  return (
   692→    <div className="p-5 overflow-y-auto">
   693→      <button
   694→        type="button"
   695→        onClick={() => setOpen((o) => !o)}
   696→        className="w-full flex items-center justify-between text-left"
   697→        aria-expanded={open}
   698→      >
   699→        <div>
   700→          <h2 className="font-serif text-lg" style={{ color: CHARCOAL }}>
   701→            Content Calendar Automation
   702→          </h2>
   703→          <p className="text-[10px]" style={{ color: "#8A8177" }}>
   704→            Goals &amp; Tracking — recurring page checks that auto-generate drafts
   705→          </p>
   706→        </div>
   707→        <span className="text-xs" style={{ color: ACCENT }}>
   708→          {open ? "▲" : "▼"}
   709→        </span>
   710→      </button>
   711→
   712→      {open && (
   713→        <div className="mt-4 space-y-3">
   714→          <div className="rounded-xl border p-3 space-y-2" style={{ borderColor: "#EDE3D6", background: "#FFFFFF" }}>
   715→            <input
   716→              value={goalTitle}
   717→              onChange={(e) => setGoalTitle(e.target.value)}
   718→              placeholder="Goal title — e.g. Weekly viral check for MIRA"
   719→              className="w-full rounded-lg border px-3 py-2 text-xs bg-white focus:outline-none"
   720→              style={{ borderColor: "#EDE3D6", color: CHARCOAL }}
   721→            />
   722→            <div className="flex gap-2">
   723→              <select
   724→                value={recurrence}
   725→                onChange={(e) => setRecurrence(e.target.value as "daily" | "weekly")}
   726→                className="rounded-lg border px-2 py-2 text-xs bg-white"
   727→                style={{ borderColor: "#EDE3D6", color: CHARCOAL }}
   728→                aria-label="Recurrence"
   729→              >
   730→                <option value="weekly">Weekly</option>
   731→                <option value="daily">Daily</option>
   732→              </select>
   733→              <select
   734→                value={goalPersonaId}
   735→                onChange={(e) => setGoalPersonaId(e.target.value)}
   736→                className="flex-1 rounded-lg border px-2 py-2 text-xs bg-white"
   737→                style={{ borderColor: "#EDE3D6", color: CHARCOAL }}
   738→                aria-label="Goal persona"
   739→              >
   740→                <option value="">Persona…</option>
   741→                {activePersona && (
   742→                  <option value={activePersona.id}>{activePersona.name}</option>
   743→                )}
   744→              </select>
   745→            </div>
   746→            <input
   747→              value={checkUrl}
   748→              onChange={(e) => setCheckUrl(e.target.value)}
   749→              placeholder="checkUrl — e.g. https://www.tiktok.com/tag/wellness"
   750→              className="w-full rounded-lg border px-3 py-2 text-xs bg-white focus:outline-none"
   751→              style={{ borderColor: "#EDE3D6", color: CHARCOAL }}
   752→            />
   753→            <button
   754→              type="button"
   755→              onClick={() => void createGoal()}
   756→              disabled={saving || !goalTitle.trim() || !checkUrl.trim() || !goalPersonaId}
   757→              className="w-full rounded-lg px-3 py-2 text-xs font-semibold text-white disabled:opacity-40"
   758→              style={{ background: ACCENT }}
   759→            >
   760→              {saving ? "Creating…" : "Create goal"}
   761→            </button>
   762→            <p className="text-[10px]" style={{ color: "#8A8177" }}>
   763→              The worker reads the page (browser worker when configured), dedupes alerts per change, and auto-writes one
   764→              in-character caption per detected change.
   765→            </p>
   766→          </div>
   767→
   768→          {goals.length > 0 && (
   769→            <div className="space-y-2">
   770→              {goals.map((g) => (
   771→                <div key={g.id} className="rounded-xl border p-3" style={{ borderColor: "#EDE3D6", background: "#FFFFFF" }}>
   772→                  <div className="flex items-start justify-between gap-2">
   773→                    <div className="min-w-0">
   774→                      <p className="text-xs font-medium truncate" style={{ color: CHARCOAL }}>
   775→                        {g.title}
   776→                      </p>
   777→                      <p className="text-[10px] mt-0.5" style={{ color: "#8A8177" }}>
   778→                        {g.recurrence} · {hostOf(g.check_url)} · checked {timeAgo(g.last_checked_at)}
   779→                        {g.failure_count ? ` · ${g.failure_count} failures` : ""}
   780→                      </p>
   781→                      <p className="text-[10px] mt-0.5" style={{ color: g.status === "active" ? "#2F6B37" : "#B3261E" }}>
   782→                        {g.status}
   783→                        {g.next_check_at ? ` · next ${timeAgo(g.next_check_at) === "just now" ? "due now" : "due soon"}` : ""}
   784→                      </p>
   785→                    </div>
   786→                    <div className="flex flex-col gap-1 shrink-0">
   787→                      <button
   788→                        type="button"
   789→                        onClick={() => void runCheck(g.id)}
   790→                        disabled={checkingId === g.id}
   791→                        className="text-[10px] rounded-md border px-2 py-1 disabled:opacity-40"
   792→                        style={{ borderColor: "#EDE3D6", color: CHARCOAL }}
   793→                      >
   794→                        {checkingId === g.id ? "…" : "Run check"}
   795→                      </button>
   796→                      <button
   797→                        type="button"
   798→                        onClick={() => void removeGoal(g.id)}
   799→                        className="text-[10px] rounded-md px-2 py-1"
   800→                        style={{ color: "#B3261E" }}
   801→                      >
   802→                        Remove
   803→                      </button>
   804→                    </div>
   805→                  </div>
   806→                </div>
   807→              ))}
   808→            </div>
   809→          )}
   810→
   811→          {alerts.length > 0 && (
   812→            <div className="space-y-1.5">
   813→              <p className="text-[10px] uppercase tracking-wider font-medium" style={{ color: "#8A8177" }}>
   814→                Recent alerts
   815→              </p>
   816→              {alerts.slice(0, 5).map((a) => (
   817→                <div key={a.id} className="rounded-lg px-2.5 py-1.5 text-[11px]" style={{ background: "#FFF3EC" }}>
   818→                  <span className="font-medium">{a.title}</span>
   819→                  <span className="block" style={{ color: "#8A8177" }}>
   820→                    {a.body}

   827→                </div>
   828→              ))}
   829→            </div>
   830→          )}
   831→        </div>
   832→      )}
   833→    </div>
   834→  );
