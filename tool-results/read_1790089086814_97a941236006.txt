    68→  const [topic, setTopic] = useState("");
    69→  const [model, setModel] = useState("openai/gpt-4o-mini");
    70→  const [variantCount, setVariantCount] = useState(3);
    71→  const [platform, setPlatform] = useState<PlatformId>("x");
    72→  const [results, setResults] = useState<VariantResult[]>([]);
    73→  const [fingerprintMeta, setFingerprintMeta] = useState<{ used: boolean; samples: number; summary: string } | null>(null);
    74→  const [loading, setLoading] = useState(false);
    75→  const [loadStage, setLoadStage] = useState(0);
    76→  const [error, setError] = useState<string | null>(null);
    77→  const [rewritingIndex, setRewritingIndex] = useState<number | null>(null);
    78→  const [copiedIndex, setCopiedIndex] = useState<number | null>(null);
    79→
    80→  // Posted-aware state
    81→  const [postedPosts, setPostedPosts] = useState<PostedPost[]>([]);
    82→  const [sensitivity, setSensitivity] = useState<Sensitivity>("medium");
    83→
    84→  // Voice DNA samples: this persona's real drafts + posted writing
    85→  const [voiceSamples, setVoiceSamples] = useState<string[]>([]);
    86→
    87→  // Vault asset context (write-for-this-asset loop)
    88→  const [assetCtx, setAssetCtx] = useState<AssetCtx | null>(null);
    89→
    90→  // Make all formats state: per result index → generated formats
    91→  const [allFormats, setAllFormats] = useState<Record<number, FormatResult[]>>({});
    92→  const [makingAll, setMakingAll] = useState<number | null>(null);
    93→
    94→  // Sticky mobile generate bar
    95→  const generateBtnRef = useRef<HTMLButtonElement>(null);
    96→  const [showStickyBar, setShowStickyBar] = useState(false);
    97→
    98→  useEffect(() => {
    99→    const stored = window.localStorage.getItem(SENSITIVITY_KEY);
   100→    if (stored === "low" || stored === "medium" || stored === "high") {
   101→      setSensitivity(stored);

   103→  }, []);
   104→
   105→  const changeSensitivity = (s: Sensitivity) => {
   106→    setSensitivity(s);
   107→    window.localStorage.setItem(SENSITIVITY_KEY, s);
   108→  };
   109→
   110→  useEffect(() => {
   111→    const load = async () => {
   112→      const {
   113→        data: { user },
   114→      } = await supabase.auth.getUser();
   115→      if (!user) {
   116→        router.push("/login");
   117→        return;
   118→      }
   119→
   120→      const { data } = await supabase
   121→        .from("personas")
   122→        .select("*")
   123→        .eq("user_id", user.id)
   124→        .order("created_at", { ascending: false });
   125→
   126→      setPersonas(data || []);
   127→      if (preselectedId) {
   128→        setSelectedId(preselectedId);
   129→        setActivePersonaId(preselectedId);
   130→      } else {
   131→        const active = getActivePersonaId();
   132→        if (active && data?.some((p) => p.id === active)) setSelectedId(active);
   133→        else if (data && data.length > 0) setSelectedId(data[0].id);
   134→      }
   135→    };
   136→    load();
   137→  }, [router, preselectedId]);
   138→
   139→  // Load posted content (repeat-avoidance) AND all drafts (voice samples)
   140→  // for the selected persona in one go.
   141→  useEffect(() => {
   142→    if (!selectedId) return;
   143→    let cancelled = false;
   144→
   145→    const load = async () => {
   146→      const [{ data: posted }, { data: drafts }] = await Promise.all([
   147→        supabase
   148→          .from("content_drafts")
   149→          .select("id, content, created_at")
   150→          .eq("persona_id", selectedId)
   151→          .eq("posted", true)
   152→          .order("created_at", { ascending: false })
   153→          .limit(20),
   154→        supabase
   155→          .from("content_drafts")
   156→          .select("content")
   157→          .eq("persona_id", selectedId)
   158→          .order("created_at", { ascending: false })
   159→          .limit(30),
   160→      ]);
   161→      if (cancelled) return;
   162→      setPostedPosts((posted || []) as PostedPost[]);
   163→      setVoiceSamples((drafts || []).map((d: { content: string }) => d.content).filter(Boolean));
   164→    };
   165→    load();
   166→
   167→    return () => {
   168→      cancelled = true;
   169→    };
   170→  }, [selectedId]);
   171→
   172→  // Client-side Voice DNA preview — instant, before any generation.
   173→  const localFingerprint = useMemo(
   174→    () => extractVoiceFingerprint(voiceSamples),
   175→    [voiceSamples]
   176→  );
   177→
   178→  // Welcome-back prefill
   179→  useEffect(() => {
   180→    if (isWelcomeBack) setTopic(WELCOME_PROMPT);
   181→  }, [isWelcomeBack]);
   182→
   183→  // Load vault asset context when arriving from "Write for this".
   184→  useEffect(() => {
   185→    if (!assetParam) return;
   186→    let cancelled = false;
   187→
   188→    const loadAsset = async () => {
   189→      const {
   190→        data: { user },
   191→      } = await supabase.auth.getUser();
   192→      if (!user) return;
   193→
   194→      const { data } = await supabase
   195→        .from("assets")
   196→        .select("id, persona_id, type, url, content, tags")
   197→        .eq("id", assetParam)
   198→        .eq("user_id", user.id)
   199→        .limit(1);
   200→
   201→      const asset = (data || [])[0] as AssetCtx | undefined;
   202→      if (!cancelled && asset) {
   203→        setAssetCtx(asset);
   204→        if (asset.persona_id) setSelectedId(asset.persona_id);
   205→      }
   206→    };
   207→    loadAsset();
   208→
   209→    return () => {
   210→      cancelled = true;
   211→    };
   212→  }, [assetParam]);
   213→
   214→  // Sticky bar: show when the main Generate button scrolls out of view
   215→  useEffect(() => {
   216→    const el = generateBtnRef.current;
   217→    if (!el) return;
   218→    const observer = new IntersectionObserver(
   219→      ([entry]) => setShowStickyBar(!entry.isIntersecting),
   220→      { threshold: 0 }
   221→    );
   222→    observer.observe(el);
   223→    return () => observer.disconnect();
   224→  }, []);
   225→
   226→  // Loading stage rotation
   227→  useEffect(() => {
   228→    if (!loading) {
   229→      setLoadStage(0);
   230→      return;
   231→    }
   232→    const stages = [
   233→      "Reading their voice…",
   234→      `Writing ${variantCount > 1 ? `${variantCount} variants` : "your draft"}…`,
   235→      "Quality pass — scrubbing clichés, checking platform fit…",
   236→      "Ranking by voice match…",
   237→    ];
   238→    const t = setInterval(() => setLoadStage((s) => (s + 1) % stages.length), 1800);
   239→    return () => clearInterval(t);
   240→  }, [loading, variantCount]);
   241→
   242→  const selectedPersona = personas.find((p) => p.id === selectedId);
   243→
   244→  // Live duplicate warning (debounced)
   245→  const [similarPosts, setSimilarPosts] = useState<SimilarPost[]>([]);
   246→  useEffect(() => {
   247→    const t = setTimeout(() => {
   248→      setSimilarPosts(findSimilarPosts(topic, postedPosts, sensitivity));
   249→    }, 350);
   250→    return () => clearTimeout(t);
   251→  }, [topic, postedPosts, sensitivity]);
   252→
   253→  const saveDraft = async (content: string, contentType: string) => {
   254→    if (!selectedPersona) return;
   255→    const {
   256→      data: { user },
   257→    } = await supabase.auth.getUser();
   258→    if (user) {
   259→      await supabase.from("content_drafts").insert({
   260→        persona_id: selectedPersona.id,
   261→        user_id: user.id,
   262→        type: contentType,
   263→        content,
   264→      });
   265→    }
   266→  };
   267→
   268→  const buildBody = (extra: Record<string, unknown> = {}) => ({
   269→    persona: selectedPersona,
   270→    type,
   271→    topic,
   272→    model,
   273→    platform,
   274→    voiceSamples,
   275→    postedContext: postedPosts,
   276→    assetContext: assetCtx
   277→      ? { type: assetCtx.type, tags: assetCtx.tags || [], content: assetCtx.content || "" }
   278→      : undefined,
   279→    ...extra,
   280→  });
   281→
   282→  const postGenerate = async (body: Record<string, unknown>) => {
   283→    const res = await fetch("/api/generate", {
   284→      method: "POST",
   285→      headers: { "Content-Type": "application/json" },
   286→      body: JSON.stringify(body),
   287→    });
   288→    const data = await res.json();
   289→    if (!res.ok) throw new Error(data.error || "Generation failed");
   290→    return data;
   291→  };
   292→
   293→  const handleGenerate = async () => {
   294→    if (!selectedPersona) return;
   295→    setLoading(true);
   296→    setError(null);
   297→    setResults([]);
   298→    setAllFormats({});
   299→
   300→    try {
   301→      const data = await postGenerate(buildBody({ variants: variantCount }));
   302→      setFingerprintMeta(data.fingerprint || null);
   303→      setResults(data.variants as VariantResult[]);
   304→      for (const v of data.variants as VariantResult[]) {
   305→        await saveDraft(v.content, type);
   306→      }
   307→      if (data.degraded) {
   308→        setError(`${data.failed} of ${variantCount} variants failed to generate — showing the ones that made it.`);
   309→      }
   310→    } catch (err: unknown) {
   311→      setError(err instanceof Error ? err.message : "Generation failed");
   312→    } finally {
   313→      setLoading(false);
   314→    }
   315→  };
   316→
   317→  // Regenerate ONE card in place (keeps the others). Rotates the structural
   318→  // strategy so "regenerate" never repeats the same architecture twice.
   319→  const handleRegenerate = async (index: number) => {
   320→    if (!selectedPersona) return;
   321→    setRewritingIndex(index);
   322→    try {
   323→      const data = await postGenerate(
   324→        buildBody({ variants: 1, strategyOffset: Math.floor(Math.random() * 4) })
   325→      );
   326→      const fresh = (data.variants as VariantResult[])[0];
   327→      if (fresh) {
   328→        const next = [...results];
   329→        next[index] = fresh;
   330→        setResults(next);
   331→        await saveDraft(fresh.content, type);
   332→      }
   333→    } catch (err: unknown) {
   334→      alert(err instanceof Error ? err.message : "Regeneration failed");
   335→    } finally {
   336→      setRewritingIndex(null);
   337→    }
   338→  };
   339→
   340→  const handleQuickRewrite = async (index: number, instruction: string) => {
   341→    if (!selectedPersona || !results[index]) return;
   342→    setRewritingIndex(index);
   343→    try {
   344→      const data = await postGenerate(
   345→        buildBody({ variants: 1, rewrite: { original: results[index].content, instruction } })
   346→      );
   347→      const fresh = (data.variants as VariantResult[])[0];
   348→      if (fresh) {
   349→        const next = [...results];
   350→        next[index] = fresh;
   351→        setResults(next);
   352→        await saveDraft(fresh.content, type);
   353→      }
   354→    } catch (err: unknown) {
   355→      alert(err instanceof Error ? err.message : "Rewrite failed");
   356→    } finally {
   357→      setRewritingIndex(null);
   358→    }
   359→  };
   360→
   361→  const handleTransform = async (index: number, newType: ContentType, instruction: string) => {
   362→    if (!selectedPersona || !results[index]) return;
   363→    setRewritingIndex(index);
   364→    try {
   365→      const res = await fetch("/api/generate", {
   366→        method: "POST",
   367→        headers: { "Content-Type": "application/json" },
   368→        body: JSON.stringify({
   369→          persona: selectedPersona,
   370→          type: newType,
   371→          model,
   372→          platform,
   373→          voiceSamples,
   374→          variants: 1,
   375→          rewrite: { original: results[index].content, instruction },
   376→        }),
   377→      });
   378→      const data = await res.json();
   379→      if (!res.ok) throw new Error(data.error || "Transform failed");
   380→      const fresh = (data.variants as VariantResult[])[0];
   381→      if (fresh) {
   382→        const next = [...results];
   383→        next[index] = { ...fresh, hookType: `→ ${newType.replace("_", " ")}` };
   384→        setResults(next);
   385→        await saveDraft(fresh.content, newType);
   386→      }
   387→    } catch (err: unknown) {
   388→      alert(err instanceof Error ? err.message : "Transform failed");
   389→    } finally {
   390→      setRewritingIndex(null);
   391→    }
   392→  };
   393→
   394→  const handleMakeAllFormats = async (index: number) => {
   395→    if (!selectedPersona || makingAll !== null || !results[index]) return;
   396→    setMakingAll(index);
   397→    setError(null);
   398→
   399→    const jobs: { label: string; type: ContentType; instruction: string }[] = [
   400→      { label: "Script", type: "script", instruction: "Turn this into a short video script (30-45 seconds) keeping the exact same voice and message" },
   401→      { label: "Caption", type: "caption", instruction: "Turn this into a strong social media caption keeping the exact same voice and message" },
   402→      { label: "Image prompt", type: "image_prompt", instruction: "Turn this into a detailed image generation prompt that matches the persona's world and this content" },
   403→    ];
   404→
   405→    try {
   406→      const formats: FormatResult[] = [];
   407→      for (const j of jobs) {
   408→        try {
   409→          const data = await postGenerate(
   410→            buildBody({ type: j.type, variants: 1, rewrite: { original: results[index].content, instruction: j.instruction } })
   411→          );
   412→          const fresh = (data.variants as VariantResult[])[0];
   413→          if (fresh) {
   414→            formats.push({ label: j.label, type: j.type, content: fresh.content });
   415→            await saveDraft(fresh.content, j.type);
   416→          }
   417→        } catch {
   418→          // one format failing shouldn't kill the others
   419→        }
   420→      }
   421→      if (formats.length === 0) throw new Error("All transforms failed");
   422→      setAllFormats((prev) => ({ ...prev, [index]: formats }));
   423→    } catch (err: unknown) {
   424→      setError(err instanceof Error ? err.message : "Transform failed");
   425→    } finally {
   426→      setMakingAll(null);
   427→    }
   428→  };
   429→
   430→  const handleCopyOnly = async (content: string, index: number) => {
   431→    const ok = await copyToClipboard(content);
   432→    if (ok) {
   433→      setCopiedIndex(index);
   434→      setTimeout(() => setCopiedIndex((c) => (c === index ? null : c)), 1500);
   435→    }
   436→  };
   437→
   438→  const sharePlatform: Record<PlatformId, Platform> = {
   439→    x: "twitter",
   440→    linkedin: "linkedin",
   441→    instagram: "instagram",
   442→    threads: "threads",
   443→  };
   444→
   445→  const generateButton = (
   446→    extraClass: string = "",
   447→    ref?: RefObject<HTMLButtonElement | null>
   448→  ) => (
   449→    <button
   450→      ref={ref}
   451→      onClick={handleGenerate}
   452→      disabled={loading || !selectedPersona}
   453→      className={`w-full min-h-[52px] bg-white text-black font-medium rounded-xl hover:bg-zinc-200 disabled:opacity-50 text-base ${extraClass}`}
   454→    >
   455→      {loading ? (
   456→        <span className="inline-flex items-center gap-2">
   457→          <span className="w-4 h-4 border-2 border-black/30 border-t-black rounded-full animate-spin" />
   458→          {["Reading their voice…", "Writing…", "Quality pass…", "Ranking…"][loadStage % 4]}
   459→        </span>
   460→      ) : (
   461→        `Generate ${variantCount > 1 ? `${variantCount} variants` : ""}`
   462→      )}
   463→    </button>
   464→  );
   465→
   466→  return (
   467→    <div className="min-h-screen p-4 sm:p-8 pb-28 sm:pb-8">
   468→      <div className="max-w-3xl mx-auto">
   469→        <div className="flex items-center justify-between mb-6 sm:mb-8">
   470→          <a href="/dashboard" className="text-sm text-zinc-400 hover:text-white">
   471→            ← Dashboard
   472→          </a>
   473→          <h1 className="text-2xl font-bold">Generate Content</h1>
   474→        </div>
   475→
   476→        {/* First-run progress banner */}
   477→        {isFirstRun && (
   478→          <div className="bg-zinc-900 border border-zinc-700 rounded-xl p-4 sm:p-5 mb-6">
   479→            <p className="text-sm font-medium text-white mb-3">Your first post, in 3 steps</p>
   480→            <ol className="space-y-2 text-sm">
   481→              <li className="flex items-center gap-2 text-green-300">
   482→                <span className="w-5 h-5 rounded-full bg-green-500/20 border border-green-600 flex items-center justify-center text-[10px]">✓</span>
   483→                Persona created
   484→              </li>
   485→              <li className="flex items-center gap-2 text-white font-medium">
   486→                <span className="w-5 h-5 rounded-full bg-white text-black flex items-center justify-center text-[10px]">2</span>
   487→                Generate your first post ← you are here
   488→              </li>
   489→              <li className="flex items-center gap-2 text-zinc-400">
   490→                <span className="w-5 h-5 rounded-full bg-zinc-800 border border-zinc-700 flex items-center justify-center text-[10px]">3</span>
   491→                Copy &amp; open X or LinkedIn
   492→              </li>
   493→            </ol>
   494→          </div>
   495→        )}
   496→
   497→        <div className="space-y-6">
   498→          <div>
   499→            <label className="block text-sm text-zinc-400 mb-2">Persona</label>
   500→            <select
   501→              value={selectedId}
   502→              onChange={(e) => {
   503→                setSelectedId(e.target.value);
   504→                setActivePersonaId(e.target.value);
   505→              }}
   506→              className="w-full px-4 py-3 min-h-[48px] bg-zinc-900 border border-zinc-700 rounded-lg"
   507→            >
   508→              {personas.map((p) => (
   509→                <option key={p.id} value={p.id}>
   510→                  {p.name}
   511→                </option>
   512→              ))}
   513→            </select>
   514→            {selectedPersona && getActivePersonaId() === selectedPersona.id && (
   515→              <p className="mt-2 text-xs text-green-400">● Active voice</p>
   516→            )}
   517→          </div>
   518→
   519→          {/* Voice DNA preview — measured from their real drafts */}
   520→          {selectedPersona && (
   521→            <div className="bg-zinc-900 border border-zinc-800 rounded-xl p-4">
   522→              <div className="flex items-center justify-between gap-3">
   523→                <div className="min-w-0">
   524→                  <p className="font-medium text-zinc-200">{selectedPersona.name}</p>
   525→                  <p className="text-xs text-zinc-500 line-clamp-1">{selectedPersona.backstory}</p>
   526→                </div>
   527→                <span
   528→                  className={`shrink-0 text-[10px] px-2 py-1 rounded-full border ${
   529→                    localFingerprint.samples >= 3
   530→                      ? "border-emerald-700 text-emerald-400 bg-emerald-950/40"
   531→                      : "border-zinc-700 text-zinc-500"
   532→                  }`}
   533→                  title={localFingerprint.samples >= 3 ? "Learned from this persona's drafts" : "Add 3+ drafts to lock the voice"}
   534→                >
   535→                  {localFingerprint.samples >= 3 ? "VOICE DNA ✓" : "VOICE DNA: LEARNING"}
   536→                </span>
   537→              </div>
   538→              {localFingerprint.samples > 0 && (
   539→                <p className="mt-2 text-[11px] text-zinc-500">
   540→                  {localFingerprint.sentenceSpread} rhythm · ~{localFingerprint.avgSentenceWords} words/sentence ·{" "}
   541→                  {localFingerprint.emojiPer100 >= 0.5 ? "uses emoji" : "no emoji"} · {localFingerprint.casing} casing
   542→                  {localFingerprint.topEmojis.length ? ` · ${localFingerprint.topEmojis.join("")}` : ""}
   543→                </p>
   544→              )}
   545→            </div>
   546→          )}
   547→
