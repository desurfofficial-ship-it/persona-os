    79→    79→function hostOf(url: string): string {
    80→    80→  try {
    81→    81→    return new URL(url).host.replace(/^www\./, "");

    87→    87→function timeAgo(iso?: string | null): string {
    88→    88→  if (!iso) return "never";
    89→    89→  const diff = Date.now() - new Date(iso).getTime();
    90→    90→  const mins = Math.floor(diff / 60000);
    91→    91→  if (mins < 1) return "just now";
    92→    92→  if (mins < 60) return `${mins}m ago`;
    93→    93→  const hours = Math.floor(mins / 60);
    94→    94→  if (hours < 24) return `${hours}h ago`;
    95→    95→  return `${Math.floor(hours / 24)}d ago`;
    96→    96→}

   105→   105→  useEffect(() => {
   106→   106→    let alive = true;
   107→   107→    (async () => {
   108→   108→      const { data } = await supabase.auth.getUser();
   109→   109→      if (!alive) return;
   110→   110→      if (!data.user) {
   111→   111→        router.push("/login");
   112→   112→        return;
   113→   113→      }
   114→   114→      setAuthReady(true);
   115→   115→      try {
   116→   116→        const res = await authedFetch("/api/personas");
   117→   117→        const json = (await res.json()) as { personas?: Persona[] };
   118→   118→        const list = Array.isArray(json.personas) ? json.personas : [];
   119→   119→        if (!alive) return;
   120→   120→        setPersonas(list);
   121→   121→        if (list.length > 0) setSelectedId((prev) => prev || list[0].id);
   122→   122→      } catch {
   123→   123→        if (alive) setPersonas([]);
   124→   124→      }
   125→   125→    })();
   126→   126→    return () => {
   127→   127→      alive = false;
   128→   128→    };
   129→   129→  }, [router]);
   130→   130→
   131→   131→  const activePersona = useMemo(
   132→   132→    () => personas.find((p) => p.id === selectedId) || null,

   139→   139→    const token = getSessionToken();
   140→   140→    return {
   141→   141→      ...(token ? { Authorization: `Bearer ${token}` } : {}),
   142→   142→      ...(selectedId ? { "x-persona-id": selectedId } : {}),
   143→   143→    };
   144→   144→  }, [selectedId]);

   146→   146→  if (!authReady) {

   157→   157→        personas={personas}
   158→   158→        selectedId={selectedId}
   159→   159→        onSelect={setSelectedId}
   160→   160→        activePersona={activePersona}

   587→   587→function GoalsPanel({

   594→   594→  refreshSignal: number;
   595→   595→}) {
   596→   596→  const [goals, setGoals] = useState<GoalRow[]>([]);
   597→   597→  const [alerts, setAlerts] = useState<AlertRow[]>([]);
   598→   598→  const [open, setOpen] = useState(false);

   701→   701→            Content Calendar Automation
   702→   702→          </h2>

   704→   704→            Goals &amp; Tracking — recurring page checks that auto-generate drafts

   709→   709→        </span>
   710→   710→      </button>
   711→   711→
   712→   712→      {open && (

   714→   714→          <div className="rounded-xl border p-3 space-y-2" style={{ borderColor: "#EDE3D6", background: "#FFFFFF" }}>
